# Configuração do cliente

Arquivo de configuração:

- **Linux/macOS:** `/etc/tungo/client_configuration.json`
- **Windows:** `%ProgramData%\TunGo\client_configuration.json`

Reinicie o cliente após fazer alterações.

## Exemplo completo {#example}

A configuração do cliente é gerada pelo servidor. Você pode personalizá-la se necessário.

```json title="client_configuration.json"
{
  "ClientID": 1, // ID do cliente atribuído pelo servidor
  "Protocol": "UDP", // Transporte ativo: TCP, UDP, WS ou WSS

  // TCPSettings — Configurações de TCP
  "TCPSettings": {
    "TunName": "c_tcptun0", // Nome da interface TUN
    "IPv4Subnet": "10.0.0.0/24", // Sub-rede IPv4 do túnel
    "IPv6Subnet": "fd00::/64", // Sub-rede IPv6 do túnel; sem este campo, o túnel não usa IPv6
    "Server": { "IPv4": "192.0.2.1" }, // Endereço do servidor; Domain ou IPv6 também podem substituir IPv4
    "Port": 8080, // Porta do servidor para conexão
    "DNSv4": ["1.1.1.1", "8.8.8.8"], // Servidores DNS IPv4; opcional, valores padrão exibidos
    "DNSv6": ["2606:4700:4700::1111", "2001:4860:4860::8888"], // Servidores DNS IPv6; opcional, valores padrão exibidos
    "MTU": 1500, // MTU do túnel em bytes
    "Protocol": "TCP", // Transporte do perfil; Protocol no nível superior seleciona o transporte ativo
    "Encryption": "ChaCha20Poly1305", // Algoritmo de criptografia
    "DialTimeoutMs": 5000 // Tempo limite de conexão em milissegundos
  },

  // UDPSettings — Configurações de UDP
  "UDPSettings": {
    "TunName": "c_udptun0", // Nome da interface TUN
    "IPv4Subnet": "10.0.1.0/24", // Sub-rede IPv4 do túnel
    "IPv6Subnet": "fd00:1::/64", // Sub-rede IPv6 do túnel; sem este campo, o túnel não usa IPv6
    "Server": { "IPv4": "192.0.2.1" }, // Endereço do servidor; Domain ou IPv6 também podem substituir IPv4
    "Port": 9090, // Porta do servidor para conexão
    "DNSv4": ["1.1.1.1", "8.8.8.8"], // Servidores DNS IPv4; opcional, valores padrão exibidos
    "DNSv6": ["2606:4700:4700::1111", "2001:4860:4860::8888"], // Servidores DNS IPv6; opcional, valores padrão exibidos
    "MTU": 1500, // MTU do túnel em bytes
    "Protocol": "UDP", // Transporte do perfil; Protocol no nível superior seleciona o transporte ativo
    "Encryption": "ChaCha20Poly1305", // Algoritmo de criptografia
    "DialTimeoutMs": 5000 // Tempo limite de conexão em milissegundos
  },

  // WSSettings — Configurações de WebSocket para WS e WSS
  "WSSettings": {
    "TunName": "c_wstun0", // Nome da interface TUN
    "IPv4Subnet": "10.0.2.0/24", // Sub-rede IPv4 do túnel
    "IPv6Subnet": "fd00:2::/64", // Sub-rede IPv6 do túnel; sem este campo, o túnel não usa IPv6
    "Server": { "IPv4": "192.0.2.1" }, // Endereço do servidor; Domain ou IPv6 também podem substituir IPv4
    "Port": 1010, // Porta do servidor para conexão
    "DNSv4": ["1.1.1.1", "8.8.8.8"], // Servidores DNS IPv4; opcional, valores padrão exibidos
    "DNSv6": ["2606:4700:4700::1111", "2001:4860:4860::8888"], // Servidores DNS IPv6; opcional, valores padrão exibidos
    "MTU": 1500, // MTU do túnel em bytes
    "Protocol": "WS", // Transporte do perfil; Protocol no nível superior seleciona o transporte ativo
    "Encryption": "ChaCha20Poly1305", // Algoritmo de criptografia
    "DialTimeoutMs": 5000 // Tempo limite de conexão em milissegundos
  },

  "X25519PublicKey": "<server-public-key>", // Chave pública do servidor da configuração gerada
  "ClientPublicKey": "<client-public-key>", // Chave pública do cliente registrada no servidor
  "ClientPrivateKey": "<client-private-key>", // Chave privada do cliente da configuração gerada

  // Sub-redes IPv4 de destino roteadas pela VPN; aqui, túnel completo
  "TunnelRoutesV4": ["0.0.0.0/1", "128.0.0.0/1"],
  // Sub-redes IPv6 de destino roteadas pela VPN; aqui, túnel completo
  "TunnelRoutesV6": ["::/1", "8000::/1"]
}
```
