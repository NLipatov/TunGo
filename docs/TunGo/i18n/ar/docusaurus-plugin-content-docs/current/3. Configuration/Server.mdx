# إعداد الخادم

ملف الإعدادات:

- **Linux:** `/etc/tungo/server_configuration.json`

أعد تشغيل الخادم بعد إجراء التغييرات.

## مثال مع تعليقات {#example}

```json title="server_configuration.json"
{
  // TCPSettings — إعدادات TCP
  "TCPSettings": {
    "TunName": "s_tcptun0", // اسم واجهة TUN
    "IPv4Subnet": "10.0.0.0/24", // شبكة النفق الفرعية IPv4
    "IPv6Subnet": "fd00::/64", // شبكة النفق الفرعية IPv6؛ بدون هذا الحقل لا يُستخدم IPv6 داخل النفق
    // الأولوية لـ IPv4 على IPv6: يستمع الخادم على 127.0.0.1 لاستقبال الاتصالات الواردة.
    "Server": {
      "IPv4": "127.0.0.1",
      "IPv6": "::1"
    },
    "Port": 8080, // منفذ الاستماع
    "MTU": 1500, // قيمة MTU للنفق بالبايت
    "Protocol": "TCP", // بروتوكول النقل
    "Encryption": "ChaCha20Poly1305", // خوارزمية التشفير
    "DialTimeoutMs": 5000 // مهلة الاتصال بالميلي ثانية
  },

  // UDPSettings — إعدادات UDP
  "UDPSettings": {
    "TunName": "s_udptun0", // اسم واجهة TUN
    "IPv4Subnet": "10.0.1.0/24", // شبكة النفق الفرعية IPv4
    "IPv6Subnet": "fd00:1::/64", // شبكة النفق الفرعية IPv6؛ بدون هذا الحقل لا يُستخدم IPv6 داخل النفق
    // يستقبل الخادم الحزم الواردة على عنوان الاسترجاع المحلي IPv6 (::1) فقط.
    "Server": { "IPv6": "::1" },
    "Port": 9090, // منفذ الاستماع
    "MTU": 1500, // قيمة MTU للنفق بالبايت
    "Protocol": "UDP", // بروتوكول النقل
    "Encryption": "ChaCha20Poly1305", // خوارزمية التشفير
    "DialTimeoutMs": 5000 // مهلة الاتصال بالميلي ثانية
  },

  // WSSettings — إعدادات WebSocket
  "WSSettings": {
    "TunName": "s_wstun0", // اسم واجهة TUN
    "IPv4Subnet": "10.0.2.0/24", // شبكة النفق الفرعية IPv4
    "IPv6Subnet": "fd00:2::/64", // شبكة النفق الفرعية IPv6؛ بدون هذا الحقل لا يُستخدم IPv6 داخل النفق
    // إذا لم يُحدَّد Server أو ضُبط على {}، يستمع الخادم على جميع واجهات الشبكة لاستقبال الاتصالات الواردة.
    // يُختار العنوان تلقائياً: "::" إذا كان IPv6 متاحاً، وإلا "0.0.0.0".
    "Port": 1010, // منفذ الاستماع
    "MTU": 1500, // قيمة MTU للنفق بالبايت
    "Protocol": "WS", // بروتوكول النقل
    "Encryption": "ChaCha20Poly1305", // خوارزمية التشفير
    "DialTimeoutMs": 5000 // مهلة الاتصال بالميلي ثانية
  },

  "Host": "", // النطاق/IP لإعدادات العملاء المُنشأة؛ فارغ للاكتشاف التلقائي
  "X25519PublicKey": "<server-public-key>", // مفتاح الخادم العام؛ يُولَّد تلقائياً عند التشغيل الأول إذا لم يكن محدداً.
  "X25519PrivateKey": "<server-private-key>", // مفتاح الخادم الخاص؛ يُولَّد تلقائياً عند التشغيل الأول إذا لم يكن محدداً.
  "ClientCounter": 1, // عدد العملاء الذين تم إنشاؤهم

  "EnableTCP": true, // true لتفعيل البروتوكول وfalse لتعطيله
  "EnableUDP": true, // true لتفعيل البروتوكول وfalse لتعطيله
  "EnableWS": true, // true لتفعيل البروتوكول وfalse لتعطيله

  // AllowedPeers — العملاء المصرّح لهم
  "AllowedPeers": [
    {
      "Name": "client-1", // تسمية اختيارية للعميل
      "PublicKey": "<client-public-key>", // مفتاح عميلك العام
      "Enabled": true, // السماح لهذا العميل بالاتصال
      "ClientID": 1 // معرّف العميل لتخصيص عناوينه في النفق
    }
  ]
}
```
