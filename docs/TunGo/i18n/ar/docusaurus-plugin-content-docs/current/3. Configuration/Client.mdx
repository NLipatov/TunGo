# إعدادات العميل

ملف الإعدادات:

- **Linux/macOS:** `/etc/tungo/client_configuration.json`
- **Windows:** `%ProgramData%\TunGo\client_configuration.json`

أعد تشغيل العميل بعد إجراء التغييرات.

## مثال كامل {#example}

ينشئ الخادم إعدادات العميل. يمكنك تخصيصها عند الحاجة.

حدّد `DialTimeout` كسلسلة نصية تتضمن وحدة الزمن: `"5s"` تعني 5 ثوانٍ، و`"500ms"` تعني 500 ميلي ثانية، و`"1m30s"` تعني دقيقة واحدة و30 ثانية. راجع [Go `time.ParseDuration`](https://pkg.go.dev/time#ParseDuration) للاطلاع على الصيغة الكاملة.

```json title="client_configuration.json"
{
  "ClientID": 1, // معرّف العميل الذي عيّنه الخادم
  "Protocol": "UDP", // بروتوكول النقل النشط: TCP أو UDP أو WS أو WSS

  // TCPSettings — إعدادات TCP
  "TCPSettings": {
    "TunName": "c_tcptun0", // اسم واجهة TUN
    "IPv4Subnet": "10.0.0.0/24", // شبكة النفق الفرعية IPv4
    "IPv6Subnet": "fd00::/64", // شبكة النفق الفرعية IPv6؛ بدون هذا الحقل لا يُستخدم IPv6 داخل النفق
    "Server": { "IPv4": "192.0.2.1" }, // عنوان الخادم؛ يمكن استخدام Domain أو IPv6 بدلاً من IPv4
    "Port": 8080, // منفذ الخادم المراد الاتصال به
    "DNSv4": ["1.1.1.1", "8.8.8.8"], // خوادم DNS لـ IPv4؛ الحقل اختياري والقيم المعروضة افتراضية
    "DNSv6": ["2606:4700:4700::1111", "2001:4860:4860::8888"], // خوادم DNS لـ IPv6؛ الحقل اختياري والقيم المعروضة افتراضية
    "MTU": 1500, // قيمة MTU للنفق بالبايت
    "Protocol": "TCP", // بروتوكول نقل هذا القسم؛ يحدد Protocol في المستوى الأعلى النقل النشط
    "Encryption": "ChaCha20Poly1305", // خوارزمية التشفير
    "DialTimeout": "5s" // مهلة الاتصال: 5 ثوانٍ
  },

  // UDPSettings — إعدادات UDP
  "UDPSettings": {
    "TunName": "c_udptun0", // اسم واجهة TUN
    "IPv4Subnet": "10.0.1.0/24", // شبكة النفق الفرعية IPv4
    "IPv6Subnet": "fd00:1::/64", // شبكة النفق الفرعية IPv6؛ بدون هذا الحقل لا يُستخدم IPv6 داخل النفق
    "Server": { "IPv4": "192.0.2.1" }, // عنوان الخادم؛ يمكن استخدام Domain أو IPv6 بدلاً من IPv4
    "Port": 9090, // منفذ الخادم المراد الاتصال به
    "DNSv4": ["1.1.1.1", "8.8.8.8"], // خوادم DNS لـ IPv4؛ الحقل اختياري والقيم المعروضة افتراضية
    "DNSv6": ["2606:4700:4700::1111", "2001:4860:4860::8888"], // خوادم DNS لـ IPv6؛ الحقل اختياري والقيم المعروضة افتراضية
    "MTU": 1500, // قيمة MTU للنفق بالبايت
    "Protocol": "UDP", // بروتوكول نقل هذا القسم؛ يحدد Protocol في المستوى الأعلى النقل النشط
    "Encryption": "ChaCha20Poly1305", // خوارزمية التشفير
    "DialTimeout": "5s" // مهلة الاتصال: 5 ثوانٍ
  },

  // WSSettings — إعدادات WebSocket لكل من WS وWSS
  "WSSettings": {
    "TunName": "c_wstun0", // اسم واجهة TUN
    "IPv4Subnet": "10.0.2.0/24", // شبكة النفق الفرعية IPv4
    "IPv6Subnet": "fd00:2::/64", // شبكة النفق الفرعية IPv6؛ بدون هذا الحقل لا يُستخدم IPv6 داخل النفق
    "Server": { "IPv4": "192.0.2.1" }, // عنوان الخادم؛ يمكن استخدام Domain أو IPv6 بدلاً من IPv4
    "Port": 1010, // منفذ الخادم المراد الاتصال به
    "DNSv4": ["1.1.1.1", "8.8.8.8"], // خوادم DNS لـ IPv4؛ الحقل اختياري والقيم المعروضة افتراضية
    "DNSv6": ["2606:4700:4700::1111", "2001:4860:4860::8888"], // خوادم DNS لـ IPv6؛ الحقل اختياري والقيم المعروضة افتراضية
    "MTU": 1500, // قيمة MTU للنفق بالبايت
    "Protocol": "WS", // بروتوكول نقل هذا القسم؛ يحدد Protocol في المستوى الأعلى النقل النشط
    "Encryption": "ChaCha20Poly1305", // خوارزمية التشفير
    "DialTimeout": "5s" // مهلة الاتصال: 5 ثوانٍ
  },

  "X25519PublicKey": "<server-public-key>", // المفتاح العام للخادم من ملف الإعدادات المُنشأ
  "ClientPublicKey": "<client-public-key>", // المفتاح العام للعميل المسجّل على الخادم
  "ClientPrivateKey": "<client-private-key>", // المفتاح الخاص للعميل من ملف الإعدادات المُنشأ

  // شبكات وجهة IPv4 التي تمر عبر VPN؛ هنا نفق كامل
  "TunnelRoutesV4": ["0.0.0.0/1", "128.0.0.0/1"],
  // شبكات وجهة IPv6 التي تمر عبر VPN؛ هنا نفق كامل
  "TunnelRoutesV6": ["::/1", "8000::/1"]
}
```
