# Конфигурация клиента

Файл конфигурации:

- **Linux/macOS:** `/etc/tungo/client_configuration.json`
- **Windows:** `%ProgramData%\TunGo\client_configuration.json`

После изменений перезапустите клиент.

## Полный пример {#example}

Клиентская конфигурация генерируется сервером. При необходимости её можно изменить.

```json title="client_configuration.json"
{
  "ClientID": 1, // ID клиента, назначенный сервером
  "Protocol": "UDP", // Активный транспорт: TCP, UDP, WS или WSS

  // TCPSettings — Настройки TCP
  "TCPSettings": {
    "TunName": "c_tcptun0", // Имя TUN-интерфейса
    "IPv4Subnet": "10.0.0.0/24", // IPv4-подсеть туннеля
    "IPv6Subnet": "fd00::/64", // IPv6-подсеть туннеля; без поля — туннель без IPv6
    "Server": { "IPv4": "192.0.2.1" }, // Адрес сервера; вместо IPv4 можно указать Domain или IPv6
    "Port": 8080, // Порт сервера для подключения
    "DNSv4": ["1.1.1.1", "8.8.8.8"], // DNS-серверы IPv4; поле необязательно, показаны значения по умолчанию
    "DNSv6": ["2606:4700:4700::1111", "2001:4860:4860::8888"], // DNS-серверы IPv6; поле необязательно, показаны значения по умолчанию
    "MTU": 1500, // MTU туннеля в байтах
    "Protocol": "TCP", // Транспорт профиля; активный транспорт выбирается верхнеуровневым Protocol
    "Encryption": "ChaCha20Poly1305", // Алгоритм шифрования
    "DialTimeoutMs": 5000 // Таймаут подключения в миллисекундах
  },

  // UDPSettings — Настройки UDP
  "UDPSettings": {
    "TunName": "c_udptun0", // Имя TUN-интерфейса
    "IPv4Subnet": "10.0.1.0/24", // IPv4-подсеть туннеля
    "IPv6Subnet": "fd00:1::/64", // IPv6-подсеть туннеля; без поля — туннель без IPv6
    "Server": { "IPv4": "192.0.2.1" }, // Адрес сервера; вместо IPv4 можно указать Domain или IPv6
    "Port": 9090, // Порт сервера для подключения
    "DNSv4": ["1.1.1.1", "8.8.8.8"], // DNS-серверы IPv4; поле необязательно, показаны значения по умолчанию
    "DNSv6": ["2606:4700:4700::1111", "2001:4860:4860::8888"], // DNS-серверы IPv6; поле необязательно, показаны значения по умолчанию
    "MTU": 1500, // MTU туннеля в байтах
    "Protocol": "UDP", // Транспорт профиля; активный транспорт выбирается верхнеуровневым Protocol
    "Encryption": "ChaCha20Poly1305", // Алгоритм шифрования
    "DialTimeoutMs": 5000 // Таймаут подключения в миллисекундах
  },

  // WSSettings — Настройки WebSocket для WS и WSS
  "WSSettings": {
    "TunName": "c_wstun0", // Имя TUN-интерфейса
    "IPv4Subnet": "10.0.2.0/24", // IPv4-подсеть туннеля
    "IPv6Subnet": "fd00:2::/64", // IPv6-подсеть туннеля; без поля — туннель без IPv6
    "Server": { "IPv4": "192.0.2.1" }, // Адрес сервера; вместо IPv4 можно указать Domain или IPv6
    "Port": 1010, // Порт сервера для подключения
    "DNSv4": ["1.1.1.1", "8.8.8.8"], // DNS-серверы IPv4; поле необязательно, показаны значения по умолчанию
    "DNSv6": ["2606:4700:4700::1111", "2001:4860:4860::8888"], // DNS-серверы IPv6; поле необязательно, показаны значения по умолчанию
    "MTU": 1500, // MTU туннеля в байтах
    "Protocol": "WS", // Транспорт профиля; активный транспорт выбирается верхнеуровневым Protocol
    "Encryption": "ChaCha20Poly1305", // Алгоритм шифрования
    "DialTimeoutMs": 5000 // Таймаут подключения в миллисекундах
  },

  "X25519PublicKey": "<server-public-key>", // Публичный ключ сервера из сгенерированного конфига
  "ClientPublicKey": "<client-public-key>", // Публичный ключ клиента, зарегистрированный на сервере
  "ClientPrivateKey": "<client-private-key>", // Приватный ключ клиента из сгенерированного конфига

  // IPv4-подсети назначения, направляемые через VPN; здесь — полный туннель
  "TunnelRoutesV4": ["0.0.0.0/1", "128.0.0.0/1"],
  // IPv6-подсети назначения, направляемые через VPN; здесь — полный туннель
  "TunnelRoutesV6": ["::/1", "8000::/1"]
}
```
