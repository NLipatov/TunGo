# Конфигурация сервера

Файл конфигурации:

- **Linux:** `/etc/tungo/server_configuration.json`

После изменений перезапустите сервер.

## Пример с комментариями {#example}

Сервер переносит `DialTimeout` в генерируемые клиентские конфигурации. Указывайте `DialTimeout` строкой с единицами времени: `"5s"` — 5 секунд, `"500ms"` — 500 миллисекунд, `"1m30s"` — 1 минута 30 секунд. Полный синтаксис описан в документации [Go `time.ParseDuration`](https://pkg.go.dev/time#ParseDuration).

```json title="server_configuration.json"
{
  // TCPSettings — Настройки TCP
  "TCPSettings": {
    "TunName": "s_tcptun0", // Имя TUN-интерфейса
    "IPv4Subnet": "10.0.0.0/24", // IPv4-подсеть туннеля
    "IPv6Subnet": "fd00::/64", // IPv6-подсеть туннеля; без поля — туннель без IPv6
    // IPv4 имеет приоритет над IPv6: сервер слушает 127.0.0.1 для приёма входящих подключений.
    "Server": {
      "IPv4": "127.0.0.1",
      "IPv6": "::1"
    },
    "Port": 8080, // Порт прослушивания
    "MTU": 1500, // MTU туннеля в байтах
    "Protocol": "TCP", // Протокол транспорта
    "Encryption": "ChaCha20Poly1305", // Алгоритм шифрования
    "DialTimeout": "5s" // Таймаут подключения: 5 секунд
  },

  // UDPSettings — Настройки UDP
  "UDPSettings": {
    "TunName": "s_udptun0", // Имя TUN-интерфейса
    "IPv4Subnet": "10.0.1.0/24", // IPv4-подсеть туннеля
    "IPv6Subnet": "fd00:1::/64", // IPv6-подсеть туннеля; без поля — туннель без IPv6
    // Сервер принимает входящие пакеты только на IPv6 loopback (::1).
    "Server": { "IPv6": "::1" },
    "Port": 9090, // Порт прослушивания
    "MTU": 1500, // MTU туннеля в байтах
    "Protocol": "UDP", // Протокол транспорта
    "Encryption": "ChaCha20Poly1305", // Алгоритм шифрования
    "DialTimeout": "5s" // Таймаут подключения: 5 секунд
  },

  // WSSettings — Настройки WebSocket
  "WSSettings": {
    "TunName": "s_wstun0", // Имя TUN-интерфейса
    "IPv4Subnet": "10.0.2.0/24", // IPv4-подсеть туннеля
    "IPv6Subnet": "fd00:2::/64", // IPv6-подсеть туннеля; без поля — туннель без IPv6
    // Если Server отсутствует или равен {}, сервер слушает все сетевые интерфейсы для приёма входящих подключений.
    // Адрес выбирается автоматически: "::", если IPv6 доступен, иначе "0.0.0.0".
    "Port": 1010, // Порт прослушивания
    "MTU": 1500, // MTU туннеля в байтах
    "Protocol": "WS", // Протокол транспорта
    "Encryption": "ChaCha20Poly1305", // Алгоритм шифрования
    "DialTimeout": "5s" // Таймаут подключения: 5 секунд
  },

  "Host": "", // Домен/IP для клиентских конфигов; пусто — автоопределение
  "X25519PublicKey": "<server-public-key>", // Публичный ключ сервера; генерируется автоматически при первом запуске, если не задан.
  "X25519PrivateKey": "<server-private-key>", // Приватный ключ сервера; генерируется автоматически при первом запуске, если не задан.
  "ClientCounter": 1, // Число созданных клиентов

  "EnableTCP": true, // true — включить протокол, false — выключить
  "EnableUDP": true, // true — включить протокол, false — выключить
  "EnableWS": true, // true — включить протокол, false — выключить

  // AllowedPeers — Разрешённые клиенты
  "AllowedPeers": [
    {
      "Name": "client-1", // Необязательная метка клиента
      "PublicKey": "<client-public-key>", // Ваш публичный ключ клиента
      "Enabled": true, // Разрешить подключение этому клиенту
      "ClientID": 1 // ID клиента для выделения адресов в туннеле
    }
  ]
}
```
